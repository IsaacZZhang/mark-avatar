## 2. 分层模型

渲染引擎的核心抽象是**图层栈**。所有元素统一成 Layer，导出时按 zIndex 顺序绘制。

```js
// engine/layer.js
// ⚠️ 坐标一律用「相对底图的归一化比例」(0~1)，不要存像素值
// 这样同一份配置在 1080 导出、540 预览、任意尺寸底图上都能正确还原

Layer = {
  id: 'layer_3',
  type: 'sticker' | 'text' | 'frame',   // 贴纸 / 文字 / 相框
  zIndex: 1,

  // 归一化变换（相对底图宽高）
  transform: {
    cx: 0.5, cy: 0.82,   // 中心点（相对底图）
    scale: 0.8,          // 宽度占底图宽的比例
    rotate: -8,          // 角度
    opacity: 0.95
  },

  // type 相关载荷
  asset: { url: 'cloud://.../sold.png', localPath: 'wxfile://...', w: 800, h: 300 },
  text:  { content: '卖掉了', font: 'bold 96px sans-serif', color: '#FF3B30', stroke: '#FFF' },

  locked: false
}
```

**为什么用归一化坐标**：底图可能是 640×640 的头像，也可能是 3000×4000 的照片，用户还会在导出时切换 1:1 / 3:4。存像素值的话每次切换都要重算，归一化则零成本。

## 3. 渲染引擎

```js
// engine/renderer.js
// 职责：把「底图 + 图层栈」画到一张 canvas 上。纯函数式，不碰页面 state。

class WatermarkRenderer {
  constructor(canvasNode) {
    this.canvas = canvasNode;
    this.ctx = canvasNode.getContext('2d');
    this.imageCache = new Map(); // url -> HTMLImageElement
  }

  /**
   * @param {Object} opts
   * @param {HTMLImageElement} opts.base   底图（已加载）
   * @param {Array<Layer>}     opts.layers 图层栈（已按 zIndex 排序）
   * @param {Object}           opts.frame  画框配置 { ratio, size }
   * @param {number}           opts.dpr
   * @param {boolean}          opts.silent 是否画小程序码角标（预览时不画）
   */
  async render({ base, layers, frame, dpr, silent = true }) {
    const { width, height } = this._resolveCanvasSize(frame, dpr);
    this.canvas.width = width;
    this.canvas.height = height;

    const ctx = this.ctx;
    ctx.clearRect(0, 0, width, height);
    ctx.save();
    ctx.scale(dpr, dpr);

    const W = width / dpr, H = height / dpr;

    // ① 底图：cover 裁剪居中
    this._drawCover(ctx, base, 0, 0, W, H);

    // ② 图层栈：逐个变换绘制
    for (const layer of layers) {
      if (layer.locked || layer.hidden) continue;
      const img = layer.type === 'text' ? null : await this._load(layer.asset);
      this._drawLayer(ctx, layer, img, W, H);
    }

    // ③ 角标：仅导出时绘制，位置固定右下，用户不可编辑
    if (!silent) {
      await this._drawBadge(ctx, W, H);
    }

    ctx.restore();
    return { width, height };
  }

  _drawLayer(ctx, layer, img, W, H) {
    const t = layer.transform;
    const px = t.cx * W, py = t.cy * H;

    let w, h;
    if (layer.type === 'text') {
      ctx.font = layer.text.font;
      w = ctx.measureText(layer.text.content).width;
      h = parseFloat(layer.text.font); // 近似行高
    } else {
      w = t.scale * W;
      h = w * (img.height / img.width);
    }

    ctx.save();
    ctx.translate(px, py);
    ctx.rotate((t.rotate * Math.PI) / 180);
    ctx.globalAlpha = t.opacity;

    if (layer.type === 'text') {
      ctx.textAlign = 'center';
      ctx.textBaseline = 'middle';
      if (layer.text.stroke) {
        ctx.lineWidth = h * 0.12;
        ctx.strokeStyle = layer.text.stroke;
        ctx.strokeText(layer.text.content, 0, 0);
      }
      ctx.fillStyle = layer.text.color;
      ctx.fillText(layer.text.content, 0, 0);
    } else {
      ctx.drawImage(img, -w / 2, -h / 2, w, h); // 以中心点绘制
    }
    ctx.restore();
  }
}
```

### 底图 cover 裁剪

```js
_drawCover(ctx, img, dx, dy, dw, dh) {
  const ir = img.width / img.height, dr = dw / dh;
  let sw, sh, sx, sy;
  if (ir > dr) {                       // 底图更宽，裁左右
    sh = img.height; sw = sh * dr;
    sx = (img.width - sw) / 2; sy = 0;
  } else {                             // 底图更高，裁上下
    sw = img.width; sh = sw / dr;
    sx = 0; sy = (img.height - sh) / 2;
  }
  ctx.drawImage(img, sx, sy, sw, sh, dx, dy, dw, dh);
}
```

## 4. 图片加载（最容易踩坑的一环）

```js
// engine/image-loader.js
const fs = wx.getFileSystemManager();

/**
 * Canvas 2D 里图片对象必须用 canvas.createImage() 创建，
 * 用 new Image() 或 wx.downloadFile 的临时路径直接赋值都会失败。
 */
function createImage(canvas, src) {
  return new Promise((resolve, reject) => {
    const img = canvas.createImage();
    img.onload = () => resolve(img);
    img.onerror = (e) => reject(new Error('image load failed: ' + src));
    img.src = src;
  });
}

/**
 * 网络图先落盘再加载：
 * 1) wx.getImageInfo 自带缓存，同一 URL 二次访问不重复下载
 * 2) 落盘后拿到本地路径，规避 canvas 对跨域/域名的限制
 */
async function loadRemote(canvas, url) {
  const cached = diskCache.get(url);
  if (cached && await exists(cached)) return createImage(canvas, cached);

  const info = await wx.getImageInfo({ src: url });   // 走微信内置缓存
  return createImage(canvas, info.path);
}
```

**关键坑位清单**（照着避）：

| 坑 | 现象 | 解法 |
|----|------|------|
| 用 `new Image()` | canvas 画不出图，无报错 | 必须 `canvas.createImage()` |
| 未做 dpr 适配 | 高清屏导出的图发虚 | `canvas.width = w * dpr` + `ctx.scale(dpr, dpr)` |
| canvas 被移出可视区 | 部分安卓机导出黑屏/空白 | 绘制期间保持 canvas 在可视区内，或用 `position:fixed` 放屏幕外但保留 1px 可见 |
| 导出时机太早 | 导出空白图 | 2d 模式无 draw 回调，但**图片 onload 未完成时导出必空**，必须 await 所有 load |
| 超大原图直接绘制 | 内存暴涨、低端机闪退 | 先用 `wx.compressImage` 压到长边 ≤ 1080 |
| `fileType: 'png'` | 相册里体积大 | 导出用 `jpg` + `quality: 0.92`，水印无透明需求 |

## 5. 导出与保存

```js
// engine/exporter.js
async function exportAndSave(canvas, { width, height, dpr }) {
  const { tempFilePath } = await wx.canvasToTempFilePath({
    canvas,                          // ← 2d 模式必须传 canvas 实例，不是 canvasId
    x: 0, y: 0, width, height,
    destWidth: width,                // 已是物理像素，不再乘 dpr
    destHeight: height,
    fileType: 'jpg',
    quality: 0.92
  });

  return saveToAlbum(tempFilePath);
}

async function saveToAlbum(filePath) {
  const setting = await wx.getSetting();
  if (setting.authSetting['scope.writePhotosAlbum'] === false) {
    // 用户曾拒绝过，直接调用会 fail，必须引导去设置页
    const res = await wx.showModal({
      title: '需要相册权限', content: '去设置页开启「保存到相册」', confirmText: '去设置'
    });
    if (res.confirm) await wx.openAppAuthorizeSetting();
    return false;
  }
  try {
    await wx.saveImageToPhotosAlbum({ filePath });
    return true;
  } catch (e) {
    if (!/auth deny|cancel/i.test(e.errMsg)) wx.showToast({ title: '保存失败', icon: 'none' });
    return false;
  }
}
```

## 6. 手势交互与性能

编辑器要"手感好"，性能上有个必须做的取舍：

| 阶段 | 画布分辨率 | 说明 |
|------|-----------|------|
| **手势进行中** | 540px（预览档） | 每帧重绘，用低分辨率保证 60fps |
| **手势结束** | 1080px（导出档） | 松手后一次性高清渲染 |
| **导出** | 1080px | 同上，复用结果 |

```js
// 双指缩放 + 旋转：从 touch 事件算变换增量
onTouchMove(e) {
  if (e.touches.length === 1) {
    // 单指拖拽：只改 cx/cy
    this._applyDrag(e.touches[0]);
  } else {
    // 双指：距离比 → scale，夹角差 → rotate
    const [a, b] = e.touches;
    const dist = Math.hypot(a.x - b.x, a.y - b.y);
    const angle = Math.atan2(b.y - a.y, b.x - a.x) * 180 / Math.PI;
    if (this._lastDist) {
      layer.transform.scale *= dist / this._lastDist;
      layer.transform.rotate += angle - this._lastAngle;
    }
    this._lastDist = dist; this._lastAngle = angle;
  }
  // 只更新 transform，不触发页面 setData（避免跨线程开销）
  this.renderer.render({ ..., dpr: PREVIEW_DPR });
}
```

**性能红线**：手势过程中**绝对不要 `setData`**。canvas 绘制在渲染层，setData 要跨 JS↔视图线程桥接，60fps 下会被打爆。只在手势结束时同步一次状态即可。

onChooseAvatar(e) {
  // e.detail.avatarUrl 是微信头像的临时文件路径（正方形）

  this.setData({ avatarUrl });
